function(dsr_target_compile_options target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE ${ARGN})
  endif()
endfunction()
function(dsr_copy_sdl2_runtime target)
  if(WIN32)
    if(NOT DSR_SDL2_RUNTIME_DLL OR NOT EXISTS "${DSR_SDL2_RUNTIME_DLL}")
      message(FATAL_ERROR "Set DSR_SDL2_RUNTIME_DLL to the matching SDL2.dll")
    endif()
    # Several executables in one directory used to run copy_if_different on the same SDL2.dll
    # concurrently, which intermittently failed ("file in use") under parallel builds. One copy
    # target per directory, built before every executable that lives there.
    string(MD5 dsr_dir_key "${CMAKE_CURRENT_BINARY_DIR}")
    set(dsr_copy_target dsr_sdl2_runtime_${dsr_dir_key})
    if(NOT TARGET ${dsr_copy_target})
      add_custom_target(${dsr_copy_target}
        COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:${target}>"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${DSR_SDL2_RUNTIME_DLL}"
                "$<TARGET_FILE_DIR:${target}>"
        VERBATIM)
    endif()
    add_dependencies(${target} ${dsr_copy_target})
  endif()
endfunction()

add_library(dsr_headers INTERFACE)
target_include_directories(dsr_headers INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/runtime/include)
target_compile_features(dsr_headers INTERFACE cxx_std_20)
if(WIN32)
  target_compile_definitions(dsr_headers INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX SDL_MAIN_HANDLED)
  if(MSVC)
    target_compile_options(dsr_headers INTERFACE /fp:strict)
  else()
    target_compile_options(dsr_headers INTERFACE -ffp-contract=off)
  endif()
  # Zig already supplies static libc++/compiler-rt; native MinGW needs this.
  if(MINGW AND NOT DSR_ZIG_TOOLCHAIN)
    add_link_options(-static -static-libgcc -static-libstdc++)
  endif()
endif()
if(DSR_RET_CHECK)
  target_compile_definitions(dsr_headers INTERFACE DSR_RET_CHECK)
endif()
if(DSR_TRACE)
  target_compile_definitions(dsr_headers INTERFACE DSR_TRACE)
endif()
