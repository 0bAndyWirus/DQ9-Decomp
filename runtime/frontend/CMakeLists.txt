if(NOT WIN32)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(SDL2 QUIET IMPORTED_TARGET sdl2)
  endif()
endif()
if(TARGET PkgConfig::SDL2)
  set(DSR_SDL2_TARGET PkgConfig::SDL2)
else()
  find_package(SDL2 REQUIRED)
  set(DSR_SDL2_TARGET SDL2::SDL2)
endif()
if(WIN32 AND NOT DSR_SDL2_RUNTIME_DLL AND TARGET SDL2::SDL2)
  foreach(config RELEASE RELWITHDEBINFO DEBUG "")
    if(config)
      get_target_property(_dsr_sdl2_runtime_dll SDL2::SDL2 IMPORTED_LOCATION_${config})
    else()
      get_target_property(_dsr_sdl2_runtime_dll SDL2::SDL2 IMPORTED_LOCATION)
    endif()
    if(_dsr_sdl2_runtime_dll MATCHES "[.][dD][lL][lL]$" AND EXISTS "${_dsr_sdl2_runtime_dll}")
      set(DSR_SDL2_RUNTIME_DLL "${_dsr_sdl2_runtime_dll}" CACHE FILEPATH
          "SDL2 runtime DLL installed alongside dq9 on Windows" FORCE)
      break()
    endif()
  endforeach()
endif()

add_library(dsr_frontend STATIC src/frontend.cpp src/frontend_logic.cpp src/ui_logic.cpp src/debug_editor.cpp src/ui_toolkit.cpp src/controller_support.cpp src/launcher.cpp src/rom_path.cpp)
target_link_libraries(dsr_frontend PUBLIC dsr_headers PRIVATE ${DSR_SDL2_TARGET})
if(WIN32)
  target_link_libraries(dsr_frontend PRIVATE shell32 ole32 winmm)
endif()
dsr_target_compile_options(dsr_frontend -Wall -Wextra -Wpedantic)
